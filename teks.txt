        }

        function toggleAdmin() {
            var box = document.getElementById('admin-box');

        function updateSistem(shouldSave = true) {
            const rates = ['usd', 'aud', 'eur', 'gbp', 'myr', 'sgd', 'cny'];
            let data = {};
            
            rates.forEach(curr => {
                data[curr] = {
                    beli: parseFloat(document.getElementById(`i-${curr}-beli`).value) || 0,
                    jual: parseFloat(document.getElementById(`i-${curr}-jual`).value) || 0
                };
                document.getElementById(`t-${curr}-beli`).innerText = data[curr].beli.toLocaleString('id-ID');
                document.getElementById(`t-${curr}-jual`).innerText = data[curr].jual.toLocaleString('id-ID');
            });

            // Menyimpan otomatis di browser HP biar data ga hilang
            if (shouldSave) {
                localStorage.setItem('ecm_rates', JSON.stringify(data));
            }

            const pilihan = document.getElementById('calc-currency').value.toLowerCase();

            const total = data[pilihan].beli * jumlah;
